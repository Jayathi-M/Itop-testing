----------------------------------------------------------------
  INSTALLATION INSTRUCTIONS
----------------------------------------------------------------

# 1. Clone the repository
    git clone https://github.com/Abhitruezenix/nexis-truezenix-app.git
    cd nexis-truezenix-app

# 2. Install frontend dependencies
    cd frontend
    npm install

# 3. Restore backend NuGet packages
    cd ../backend
    dotnet restore

# 4. Run frontend (development)
    cd frontend
    npm run dev
    # Runs at: http://localhost:5173

# 5. Run backend (development)
    cd backend
    dotnet run
    # Runs at: https://localhost:7000
    # Swagger UI: https://localhost:7000/swagger

----------------------------------------------------------------
  ENVIRONMENT VARIABLES
----------------------------------------------------------------

# Frontend (.env in /frontend)
    VITE_API_BASE_URL=https://localhost:7000/api

# Backend (appsettings.Development.json in /backend)
    ConnectionStrings__DefaultConnection=server=localhost;database=nexis_db;user=root;password=<your_password>
    JwtSettings__SecretKey=<your_jwt_secret>
    JwtSettings__Issuer=XceptionAI
    JwtSettings__Audience=XceptionAIClient
